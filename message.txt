// ===================== SAQLASH (SharedPreferences) =====================

object Store {
    private fun sp(c: Context) = c.getSharedPreferences("edu_prefs", Context.MODE_PRIVATE)

    private fun users(c: Context): JSONArray =
        try {
            JSONArray(sp(c).getString("users", "[]") ?: "[]")
        } catch (e: Exception) {
            JSONArray()
        }

    fun findUser(c: Context, contact: String): JSONObject? {
        val a = users(c)
        for (i in 0 until a.length()) {
            val o = a.optJSONObject(i) ?: continue
            if (o.optString("contact") == contact) return o
        }
        return null
    }

    fun addUser(c: Context, name: String, contact: String, hash: String): Boolean {
        return try {
            val a = users(c)
            a.put(JSONObject().put("name", name).put("contact", contact).put("hash", hash))
            sp(c).edit().putString("users", a.toString()).commit()
        } catch (e: Exception) {
            false
        }
    }

    fun themeMode(c: Context): Int {
        val m = sp(c).getInt("theme_mode", 0)
        return if (m in 0..2) m else 0
    }

    fun setThemeMode(c: Context, m: Int) {
        sp(c).edit().putInt("theme_mode", m).apply()
    }

    fun session(c: Context): String = sp(c).getString("session", "") ?: ""

    fun setSession(c: Context, contact: String) {
        sp(c).edit().putString("session", contact).apply()
    }

    fun done(c: Context, contact: String): Set<String> =
        (sp(c).getString("done_" + contact, "") ?: "").split(",").filter { it.isNotBlank() }.toSet()

    fun markDone(c: Context, contact: String, lessonId: String) {
        val s = done(c, contact).toMutableSet()
        s.add(lessonId)
        sp(c).edit().putString("done_" + contact, s.joinToString(",")).apply()
    }

    fun setRecent(c: Context, contact: String, type: String, id: String) {
        try {
            val o = JSONObject().put("type", type).put("id", id).put("ts", System.currentTimeMillis())
            sp(c).edit().putString("recent_" + contact, o.toString()).apply()
        } catch (e: Exception) {
            // e'tiborsiz qoldiriladi
        }
    }

    fun recent(c: Context, contact: String): JSONObject? =
        try {
            val s = sp(c).getString("recent_" + contact, null)
            if (s == null) null else JSONObject(s)
        } catch (e: Exception) {
            null
        }

    fun results(c: Context, contact: String): List<JSONObject> {
        return try {
            val a = JSONArray(sp(c).getString("results_" + contact, "[]") ?: "[]")
            val list = ArrayList<JSONObject>()
            for (i in 0 until a.length()) {
                val o = a.optJSONObject(i)
                if (o != null) list.add(o)
            }
            list
        } catch (e: Exception) {
            emptyList()
        }
    }

    fun addResult(c: Context, contact: String, courseId: String, score: Int, total: Int) {
        try {
            val list = results(c, contact).toMutableList()
            list.add(
                JSONObject().put("c", courseId).put("s", score).put("t", total)
                    .put("ts", System.currentTimeMillis())
            )
            val trimmed = if (list.size > 50) list.subList(list.size - 50, list.size) else list
            val arr = JSONArray()
            for (o in trimmed) arr.put(o)
            sp(c).edit().putString("results_" + contact, arr.toString()).apply()
        } catch (e: Exception) {
            // e'tiborsiz qoldiriladi
        }
    }
}

// ===================== YORDAMCHI FUNKSIYALAR =====================

fun sha256(s: String): String =
    MessageDigest.getInstance("SHA-256").digest(s.toByteArray(Charsets.UTF_8))
        .joinToString("") { "%02x".format(it) }

fun normContact(raw: String): String {
    val t = raw.trim()
    return if (t.contains("@")) t.lowercase() else t.replace(Regex("[\\s\\-()]"), "")
}

fun validContact(c: String): Boolean {
    if (c.isEmpty()) return false
    return if (c.contains("@")) {
        Patterns.EMAIL_ADDRESS.matcher(c).matches()
    } else {
        Regex("^\\+?[0-9]{9,13}\$").matches(c)
    }
}

fun fmt(ts: Long): String =
    try {
        SimpleDateFormat("dd.MM.yyyy HH:mm", Locale.getDefault()).format(Date(ts))
    } catch (e: Exception) {
        ""
    }

fun courseDone(course: Course, done: Set<String>): Int = course.lessons.count { it.id in done }

fun percentOf(r: JSONObject): Int {
    val t = r.optInt("t")
    return if (t <= 0) 0 else r.optInt("s") * 100 / t
}

@Composable
fun EduTheme(mode: Int, content: @Composable () -> Unit) {
    val sysDark = isSystemInDarkTheme()
    val dark = when (mode) {
        1 -> false
        2 -> true
        else -> sysDark
    }
    val t = if (dark) DarkEdu else LightEdu
    val bg by animateColorAsState(t.bg, animationSpec = tween(350), label = "bg")
    val surface by animateColorAsState(t.surface, animationSpec = tween(350), label = "surface")
    val surfaceHi by animateColorAsState(t.surfaceHi, animationSpec = tween(350), label = "surfaceHi")
    val text by animateColorAsState(t.text, animationSpec = tween(350), label = "text")
    val textSub by animateColorAsState(t.textSub, animationSpec = tween(350), label = "textSub")
    val line by animateColorAsState(t.line, animationSpec = tween(350), label = "line")
    val edu = EduColors(bg, surface, surfaceHi, text, textSub, line, dark)

    val scheme = if (dark) {
        darkColorScheme(
            primary = Brand1, onPrimary = OnBrand,
            background = edu.bg, onBackground = edu.text,
            surface = edu.surface, onSurface = edu.text,

@Composable
fun Appear(delayMs: Int = 0, content: @Composable () -> Unit) {
    var shown by remember { mutableStateOf(false) }
    LaunchedEffect(Unit) {
        delay(delayMs.toLong())
        shown = true
    }
    AnimatedVisibility(
        visible = shown,
        enter = fadeIn(animationSpec = tween(350)) +
                slideInVertically(animationSpec = tween(350)) { h -> h / 8 }
    ) {
        content()
    }
}

@Composable
fun Modifier.scaleClick(onClick: () -> Unit): Modifier {
    val src = remember { MutableInteractionSource() }
    val pressed by src.collectIsPressedAsState()
    val s by animateFloatAsState(if (pressed) 0.96f else 1f, animationSpec = tween(120), label = "press")
    return this
        .graphicsLayer {
            scaleX = s
            scaleY = s
        }
        .clickable(interactionSource = src, indication = null, onClick = onClick)
}

@Composable
fun Bar(f: Float, fill: Brush, track: Color, height: Dp = 8.dp) {
    var target by remember { mutableStateOf(0f) }
    LaunchedEffect(f) { target = f.coerceIn(0f, 1f) }
    val a by animateFloatAsState(target, animationSpec = tween(800), label = "bar")
    Box(
        modifier = Modifier
            .fillMaxWidth()
            .height(height)
            .clip(RoundedCornerShape(height))
            .background(track)
    ) {
        Box(
            modifier = Modifier
                .fillMaxWidth(a)
                .fillMaxHeight()
                .clip(RoundedCornerShape(height))
                .background(fill)
        )
    }
}

@Composable
fun ProgressRing(
    progress: Float,
    diameter: Dp,
    stroke: Dp,
    track: Color,
    colors: List<Color>,
    label: String,
    labelColor: Color,
    labelSize: Int = 20
) {
    var target by remember { mutableStateOf(0f) }
    LaunchedEffect(progress) { target = progress.coerceIn(0f, 1f) }
    val a by animateFloatAsState(target, animationSpec = tween(900), label = "ring")
    Box(modifier = Modifier.size(diameter), contentAlignment = Alignment.Center) {
        Canvas(modifier = Modifier.fillMaxSize()) {
            val sw = stroke.toPx()
            val tl = Offset(sw / 2f, sw / 2f)
            val sz = Size(this.size.width - sw, this.size.height - sw)
            drawArc(
                color = track,
                startAngle = -90f,
                sweepAngle = 360f,
                useCenter = false,
                topLeft = tl,
                size = sz,
                style = Stroke(width = sw, cap = StrokeCap.Round)
            )
            if (a > 0.001f) {
                drawArc(
                    brush = Brush.linearGradient(
                        colors,
                        start = Offset(0f, 0f),
                        end = Offset(this.size.width, this.size.height)
                    ),
                    startAngle = -90f,
                    sweepAngle = 360f * a,
                    useCenter = false,
                    topLeft = tl,
                    size = sz,
                    style = Stroke(width = sw, cap = StrokeCap.Round)
                )
            }
        }
        Text(label, color = labelColor, fontWeight = FontWeight.ExtraBold, fontSize = labelSize.sp)
    }
}

@Composable
fun EduApp(themeMode: Int, onTheme: (Int) -> Unit) {
    val ctx = LocalContext.current
    var contact by rememberSaveable { mutableStateOf(Store.session(ctx)) }
    val valid = contact.isNotBlank() && Store.findUser(ctx, contact) != null
    if (valid) {
        MainShell(
            contact = contact,
            themeMode = themeMode,

            onLogout = {

            }
        )
    } else {
        AuthScreen(onAuth = { c ->
            Store.setSession(ctx, c)
            contact = c
        })
    }
}

@Composable
fun AuthScreen(onAuth: (String) -> Unit) {
    val ctx = LocalContext.current
    val edu = LocalEdu.current
    var register by rememberSaveable { mutableStateOf(false) }
    var name by rememberSaveable { mutableStateOf("") }
    var contactIn by rememberSaveable { mutableStateOf("") }
    var password by rememberSaveable { mutableStateOf("") }
    var showPw by rememberSaveable { mutableStateOf(false) }
    var nameErr by rememberSaveable { mutableStateOf("") }
    var contactErr by rememberSaveable { mutableStateOf("") }
    var pwErr by rememberSaveable { mutableStateOf("") }
    var formErr by rememberSaveable { mutableStateOf("") }
    var notFound by rememberSaveable { mutableStateOf(false) }
    var busy by rememberSaveable { mutableStateOf(false) }

    fun clearErrors() {
        nameErr = ""
        contactErr = ""
        pwErr = ""
        formErr = ""
        notFound = false
    }

    fun validate(): Boolean {
        clearErrors()
        val c = normContact(contactIn)
        var ok = true
        if (register && name.trim().length < 2) {
            nameErr = "Ism kamida 2 ta belgidan iborat bo'lsin."
            ok = false
        }
        if (!validContact(c)) {
            contactErr = "Elektron pochta yoki telefon noto'g'ri. Masalan: ali@mail.com yoki +998901234567"
            ok = false
        }
        if (password.length < 6) {
            pwErr = "Parol kamida 6 ta belgidan iborat bo'lsin."
            ok = false
        }
        return ok
    }

    fun perform() {
        val c = normContact(contactIn)
        try {
            if (register) {
                if (Store.findUser(ctx, c) != null) {
                    contactErr = "Bu hisob allaqachon mavjud. Kirish rejimiga o'ting."
                } else if (Store.addUser(ctx, name.trim(), c, sha256(c + password))) {
                    onAuth(c)
                } else {
                    formErr = "Saqlashda xatolik yuz berdi. Qayta urinib ko'ring."
                }
            } else {
                val u = Store.findUser(ctx, c)
                if (u == null) {
                    notFound = true
                    contactErr = "Bu hisob topilmadi."
                } else if (u.optString("hash") != sha256(c + password)) {
                    pwErr = "Parol noto'g'ri."
                } else {
                    onAuth(c)
                }
            }
        } catch (e: Exception) {
            formErr = "Kutilmagan xatolik yuz berdi. Qayta urinib ko'ring."
        }
    }

    LaunchedEffect(busy) {
        if (busy) {
            delay(500)
            perform()
            busy = false
        }
    }

    val fieldColors = OutlinedTextFieldDefaults.colors(
        focusedBorderColor = Brand1,
        unfocusedBorderColor = edu.line,
        errorBorderColor = BadRed,
        focusedLabelColor = Brand1,
        unfocusedLabelColor = edu.textSub,
        errorLabelColor = BadRed,
        cursorColor = Brand1,
        focusedTextColor = edu.text,
        unfocusedTextColor = edu.text,
        errorTextColor = edu.text
    )
    val fieldShape = RoundedCornerShape(16.dp)

    Scaffold(
        containerColor = edu.bg,
        contentColor = edu.text,
        contentWindowInsets = WindowInsets(0.dp, 0.dp, 0.dp, 0.dp)
    ) { pad ->
        Column(
            modifier = Modifier
                .padding(pad)
                .fillMaxSize()
                .imePadding()
                .verticalScroll(rememberScrollState())
        ) {
            Box(
                modifier = Modifier
                    .fillMaxWidth()
                    .clip(RoundedCornerShape(bottomStart = 40.dp, bottomEnd = 40.dp))
                    .background(Brush.linearGradient(BrandColors))
            ) {
                Column(
                    modifier = Modifier
                        .fillMaxWidth()
                        .statusBarsPadding()
                        .padding(top = 24.dp, bottom = 64.dp),
                    horizontalAlignment = Alignment.CenterHorizontally
                ) {
                    Box(
                        modifier = Modifier
                            .size(88.dp)
                            .clip(CircleShape)
                            .background(OnBrand.copy(alpha = 0.22f)),
                        contentAlignment = Alignment.Center
                    ) {
                        Text("🎓", fontSize = 46.sp)
                    }
                    Spacer(Modifier.height(12.dp))
                    Text(
                        "EduPlatform",
                        color = OnBrand,
                        fontSize = 30.sp,
                        fontWeight = FontWeight.ExtraBold,
                        letterSpacing = 0.5.sp
                    )
                    Text(
                        "Abituriyentlar uchun ta'lim ilovasi",
                        color = OnBrand.copy(alpha = 0.88f),
                        fontSize = 14.sp,
                        textAlign = TextAlign.Center
                    )
                }
            }

            Column(
                modifier = Modifier
                    .padding(horizontal = 20.dp)
                    .offset(y = (-36).dp)
            ) {
                EduCard {
                    Row(
                        modifier = Modifier
                            .fillMaxWidth()
                            .clip(RoundedCornerShape(16.dp))
                            .background(edu.surfaceHi)
                            .padding(4.dp)
                    ) {
                        val tabShape = RoundedCornerShape(12.dp)
                        listOf(false to "Kirish", true to "Ro'yxatdan o'tish").forEach { item ->
                            val sel = register == item.first
                            val tabBg = if (sel) {
                                Modifier.background(Brush.horizontalGradient(BrandColors))

                            onFinish = {
                                Store.markDone(ctx, contact, lp.second.id)
                                ver++
                            },
                            onNext = openLesson,
                            onBack = { overlay = "course:" + lp.first.id }
                        )
                        cp != null -> CourseScreen(
                            course = cp,
                            done = done,
                            results = results,
                            onBack = { overlay = "" },
                            onLesson = openLesson,
                            onQuiz = openQuiz
                        )
                        qc != null -> QuizScreen(
                            course = qc,
                            onBack = { overlay = "course:" + qc.id },
                            onFinish = { score, total ->
                                Store.addResult(ctx, contact, qc.id, score, total)
                                ver++
                            }
                        )
                        k == "tab1" -> CoursesScreen(done = done, onCourse = openCourse)
                        k == "tab2" -> ProfileScreen(
                            name = name,
                            contact = contact,
                            done = done,
                            results = results,
                            themeMode = themeMode,
                            onTheme = onTheme,
                            onLogout = onLogout
                        )
                        else -> HomeScreen(
                            name = name,
                            done = done,
                            recent = recent,
                            onCourse = openCourse,
                            onLesson = openLesson,
                            onQuiz = openQuiz
                        )
                    }
                }
            }
        }
    }
}

// Qaytaradi: Triple(tur, id, sarlavha) yoki null
fun continueTarget(recent: JSONObject?): Triple<String, String, String>? {
    if (recent == null) return null
    val type = recent.optString("type")
    val id = recent.optString("id")
    if (type == "lesson") {
        val lp = findLesson(id)
        if (lp != null) return Triple("lesson", id, "📖 " + lp.second.title)
    } else if (type == "quiz") {
        val c = findCourse(id)
        if (c != null) return Triple("quiz", id, "📝 " + c.title + " — yakuniy test")
    }
    return null
}

@Composable
fun QuizScreen(course: Course, onBack: () -> Unit, onFinish: (Int, Int) -> Unit) {
    val edu = LocalEdu.current
    val cols = courseColors(course.id, edu.dark)
    val qs = course.questions
    val n = qs.size
    var index by rememberSaveable { mutableStateOf(0) }
    var answers by rememberSaveable { mutableStateOf(IntArray(n) { -1 }) }
    var finished by rememberSaveable { mutableStateOf(false) }
    var warning by rememberSaveable { mutableStateOf("") }

    if (n == 0 || answers.size != n) {
        Screen {
            BackLink("← Kursga qaytish", Brand1, onBack)
            Text("Bu fan uchun test hozircha mavjud emas.")
        }
        return
    }

    val idx = index.coerceIn(0, n - 1)

    if (finished) {
        val score = qs.indices.count { answers[it] == qs[it].correct }
        val frac = score.toFloat() / n
        val pct = score * 100 / n
        Screen {
            Appear(0) {
                Column(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalAlignment = Alignment.CenterHorizontally
                ) {
                    Text(course.title + " — natija", color = edu.textSub, fontSize = 14.sp)
                    Spacer(Modifier.height(12.dp))
                    ProgressRing(
                        progress = frac,
                        diameter = 160.dp,
                        stroke = 16.dp,
                        track = edu.line,
                        colors = cols,
                        label = score.toString() + "/" + n,
                        labelColor = edu.text,
                        labelSize = 32
                    )
                    Spacer(Modifier.height(12.dp))
                    Text("Foiz: " + pct + "%", fontSize = 20.sp, fontWeight = FontWeight.Bold)
                    Spacer(Modifier.height(4.dp))
                    Text(
                        if (pct >= 60) "🎉 Ajoyib!" else "💪 Yana urinib ko'ring — albatta uddalaysiz!",
                        fontSize = 18.sp,
                        fontWeight = FontWeight.ExtraBold,
                        textAlign = TextAlign.Center
                    )
                }
            }
            Text("Savollar tahlili", fontSize = 20.sp, fontWeight = FontWeight.SemiBold)
            qs.forEachIndexed { i, qu ->
                val a = answers[i]
                val ok = a == qu.correct
                val mark = if (ok) OkGreen else BadRed
                EduCard(pad = 0.dp) {
                    Row(modifier = Modifier.height(IntrinsicSize.Min)) {
                        Box(modifier = Modifier.width(6.dp).fillMaxHeight().background(mark))
                        Column(modifier = Modifier.padding(16.dp)) {
                            Text(
                                (i + 1).toString() + ". " + qu.text,
                                fontWeight = FontWeight.SemiBold,
                                fontSize = 16.sp,
                                lineHeight = 22.sp
                            )
                            Spacer(Modifier.height(6.dp))
                            Text(
                                (if (ok) "✓ To'g'ri: " else "✗ Noto'g'ri: ") + (qu.options.getOrNull(a) ?: "—"),
                                color = mark,
                                fontWeight = FontWeight.SemiBold
                            )
                            if (!ok) {
                                Text(
                                    "To'g'ri javob: " + (qu.options.getOrNull(qu.correct) ?: ""),
                                    color = OkGreen,
                                    fontWeight = FontWeight.SemiBold
                                )
                            }
                            Spacer(Modifier.height(6.dp))
                            Text(qu.explanation, color = edu.textSub, fontSize = 13.sp, lineHeight = 19.sp)
                        }
                    }
                }
            }
            GradButton(
                text = "Qayta urinish",
                colors = cols,
                onClick = {
                    answers = IntArray(n) { -1 }
                    index = 0
                    warning = ""
                    finished = false
                }
            )
            OutlineButton(text = "Kursga qaytish", onClick = onBack, color = edu.textSub)
        }
    } else {
        val qu = qs[idx]
        val letters = listOf("A", "B", "C", "D", "E", "F")
        Screen {
            BackLink("← Kursga qaytish", Brand1, onBack)
            Text(
                "Savol " + (idx + 1) + " / " + n,
                color = edu.textSub,
                fontSize = 13.sp,
                fontWeight = FontWeight.SemiBold
            )
            Row(modifier = Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.spacedBy(4.dp)) {
                for (i in 0 until n) {
                    Seg(active = i <= idx, color = cols[0], track = edu.line, modifier = Modifier.weight(1f))
                }
            }
            Text(qu.text, fontSize = 22.sp, lineHeight = 30.sp, fontWeight = FontWeight.ExtraBold)
            qu.options.forEachIndexed { i, opt ->
                OptionCard(
                    letter = letters.getOrElse(i) { "•" },
                    text = opt,
                    selected = answers[idx] == i,
                    accent = cols[0],
                    onClick = {
                        val copy = answers.copyOf()
                        copy[idx] = i
                        answers = copy
                        warning = ""
                    }
                )
            }
            if (warning.isNotEmpty()) {
                Text("⚠️ " + warning, color = BadRed, fontSize = 14.sp, fontWeight = FontWeight.SemiBold)
            }
            Row(modifier = Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.spacedBy(12.dp)) {
                if (idx > 0) {
                    OutlineButton(
                        text = "Oldingi",
                        color = edu.textSub,
                        modifier = Modifier.weight(1f),
                        onClick = {
                            warning = ""
                            index = (idx - 1).coerceAtLeast(0)
                        }
                    )
                }
                if (idx < n - 1) {
                    GradButton(
                        text = "Keyingi",
                        colors = cols,
                        modifier = Modifier.weight(1f),
                        onClick = {
                            if (answers[idx] < 0) {
                                warning = "Iltimos, avval javobni tanlang."
                            } else {
                                warning = ""
                                index = idx + 1
                            }
                        }
                    )
                } else {
                    GradButton(
                        text = "Yakunlash",
                        colors = cols,
                        modifier = Modifier.weight(1f),
                        onClick = {
                            val unanswered = answers.count { it < 0 }
                            if (answers[idx] < 0) {
                                warning = "Iltimos, avval javobni tanlang."
                            } else if (unanswered > 0) {
                                warning = "Yana " + unanswered + " ta savolga javob berilmagan."
                            } else {
                                val score = qs.indices.count { answers[it] == qs[it].correct }
                                warning = ""
                                finished = true
                                onFinish(score, n)
                            }
                        }
                    )
                }
            }
        }
    }
}
